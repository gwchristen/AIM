        private string GetDefaultInstallPath()
        {
            return Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ProgramFiles), "AIM");
        }

        private void BrowseFolder(TextBox targetTextBox)
        {
            using (var dialog = new System.Windows.Forms.FolderBrowserDialog())
            {
                dialog.Description = "Select installation folder";
                dialog.SelectedPath = targetTextBox.Text;
                if (dialog.ShowDialog() == DialogResult.OK)
                {
                    targetTextBox.Text = dialog.SelectedPath;
                }
            }
        }

        private async Task InstallAsync(TextBox logTextBox, Label stepLabel, Panel pathPanel, Panel configPanel)
        {
            try
            {
                logTextBox.Visible = true;
                pathPanel.Enabled = false;
                configPanel.Enabled = false;

                Log(logTextBox, "Starting AIM installation...");
                Log(logTextBox, $"Installation path: {_extractPath}");
                Log(logTextBox, "");

                // Step 1: Extract portable app
                Log(logTextBox, "Step 1/3: Extracting application files...");
                await Task.Delay(100); // Let UI update
                ExtractPortableApp();
                Log(logTextBox, "✓ Application files extracted");
                Log(logTextBox, "");

                // Step 2: Import security config
                Log(logTextBox, "Step 2/3: Setting up security configuration...");
                await Task.Delay(100);
                var passphraseForm = new PassphrasePrompt(_sharedConfigPath);
                if (passphraseForm.ShowDialog() == DialogResult.OK)
                {
                    await ImportSecurityConfigAsync(logTextBox, passphraseForm.Passphrase);
                    Log(logTextBox, "✓ Security configuration imported");
                }
                else
                {
                    Log(logTextBox, "⚠ Security configuration skipped (you can run import-security.ps1 later)");
                }
                Log(logTextBox, "");

                // Step 3: Create shortcuts
                Log(logTextBox, "Step 3/3: Creating shortcuts...");
                await Task.Delay(100);
                CreateShortcuts();
                Log(logTextBox, "✓ Shortcuts created");
                Log(logTextBox, "");

                Log(logTextBox, "========================================");
                Log(logTextBox, "Installation Complete!");
                Log(logTextBox, "========================================");
                Log(logTextBox, "");
                Log(logTextBox, $"AIM has been installed to: {_extractPath}");
                Log(logTextBox, "");
                Log(logTextBox, "You can now launch AIM from:");
                Log(logTextBox, "  • Start Menu → AIM");
                Log(logTextBox, $"  • Or directly: {Path.Combine(_extractPath, "AIM.exe")}");

                _installationComplete = true;

                // Ask to launch app
                var result = MessageBox.Show(
                    "Installation complete. Would you like to launch AIM now?",
                    "Installation Complete",
                    MessageBoxButtons.YesNo,
                    MessageBoxIcon.Information);

                if (result == DialogResult.Yes)
                {
                    LaunchApp();
                }

                this.Close();
            }
            catch (Exception ex)
            {
                Log(logTextBox, "");
                Log(logTextBox, $"ERROR: {ex.Message}");
                Log(logTextBox, ex.StackTrace);
                MessageBox.Show($"Installation failed: {ex.Message}", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void ExtractPortableApp()
        {
            // Get the embedded portable app (this would be bundled with the installer)
            // For now, we assume the portable files are in a subdirectory or embedded resource
            string sourceAppDir = Path.Combine(AppContext.BaseDirectory, "app");

            if (!Directory.Exists(sourceAppDir))
            {
                throw new DirectoryNotFoundException($"Portable app files not found at: {sourceAppDir}");
            }

            // Create installation directory
            if (!Directory.Exists(_extractPath))
            {
                Directory.CreateDirectory(_extractPath);
            }

            // Copy all files recursively
            CopyDirectory(sourceAppDir, _extractPath, true);
        }

        private void CopyDirectory(string sourceDir, string destDir, bool copySubDirs)
        {
            var dir = new DirectoryInfo(sourceDir);
            if (!dir.Exists)
                throw new DirectoryNotFoundException($"Source directory does not exist or could not be found: {sourceDir}");

            if (!Directory.Exists(destDir))
                Directory.CreateDirectory(destDir);

            FileInfo[] files = dir.GetFiles();
            foreach (FileInfo file in files)
            {
                string tempPath = Path.Combine(destDir, file.Name);
                file.CopyTo(tempPath, true);
            }

            if (copySubDirs)
            {
                DirectoryInfo[] dirs = dir.GetDirectories();
                foreach (DirectoryInfo subdir in dirs)
                {
                    string tempPath = Path.Combine(destDir, subdir.Name);
                    CopyDirectory(subdir.FullName, tempPath, copySubDirs);
                }
            }
        }

        private async Task ImportSecurityConfigAsync(TextBox logTextBox, string passphrase)
        {
            try
            {
                Log(logTextBox, "Connecting to shared config...");

                if (!File.Exists(_sharedConfigPath))
                {
                    Log(logTextBox, "⚠ Shared config file not accessible. You can import it later using:");
                    Log(logTextBox, $"  .\\tools\\import-security.ps1 -SharedPath \"{_sharedConfigPath}\" -AskPass");
                    return;
                }

                // Call the ImportSharedSecurity helper (if available)
                string helperPath = Path.Combine(_extractPath, "tools", "ImportSharedSecurity.exe");
                string localConfigPath = Path.Combine(
                    Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                    "AIM", "Security", "security.config");

                if (!File.Exists(helperPath))
                {
                    Log(logTextBox, "⚠ Security helper not found. Manual import required.");
                    return;
                }

                Log(logTextBox, "Running security import helper...");

                var process = new Process
                {
                    StartInfo = new ProcessStartInfo
                    {
                        FileName = helperPath,
                        Arguments = $"--import --in \"{_sharedConfigPath}\" --out \"{localConfigPath}\" --pass \"{passphrase}\"",
                        UseShellExecute = false,
                        RedirectStandardOutput = true,
                        RedirectStandardError = true,
                        CreateNoWindow = true
                    }
                };

                process.Start();
                await process.WaitForExitAsync();

                if (process.ExitCode != 0)
                {
                    string error = process.StandardError.ReadToEnd();
                    throw new Exception($"Security import failed: {error}");
                }

                Log(logTextBox, "Security config imported successfully");
            }
            catch (Exception ex)
            {
                Log(logTextBox, $"⚠ Warning: Could not import security config: {ex.Message}");
                Log(logTextBox, "  You can import it later by running: .\\tools\\import-security.ps1");
            }
        }

        private void CreateShortcuts()
        {
            try
            {
                string aimExe = Path.Combine(_extractPath, "AIM.exe");
                string startMenuFolder = Path.Combine(
                    Environment.GetFolderPath(Environment.SpecialFolder.StartMenu),
                    "Programs", "AIM");

                // Create Start Menu folder
                if (!Directory.Exists(startMenuFolder))
                {
                    Directory.CreateDirectory(startMenuFolder);
                }

                // Create shortcut using Windows Script Host
                string lnkPath = Path.Combine(startMenuFolder, "AIM.lnk");
                CreateWindowsShortcut(aimExe, lnkPath, "AIM - Asset Inventory Management");

                // Create Desktop shortcut (optional)
                string desktopFolder = Environment.GetFolderPath(Environment.SpecialFolder.Desktop);
                string desktopLnk = Path.Combine(desktopFolder, "AIM.lnk");
                CreateWindowsShortcut(aimExe, desktopLnk, "AIM - Asset Inventory Management");
            }
            catch (Exception ex)
            {
                // Non-fatal error
                Debug.WriteLine($"Failed to create shortcuts: {ex.Message}");
            }
        }

        private void CreateWindowsShortcut(string targetPath, string shortcutPath, string description)
        {
            dynamic shell = Activator.CreateInstance(Type.GetTypeFromProgID("WScript.Shell"));
            dynamic shortcut = shell.CreateShortcut(shortcutPath);
            shortcut.TargetPath = targetPath;
            shortcut.WorkingDirectory = Path.GetDirectoryName(targetPath);
            shortcut.Description = description;
            shortcut.IconLocation = targetPath + ",0";
            shortcut.Save();

            System.Runtime.InteropServices.Marshal.FinalReleaseComObject(shortcut);
            System.Runtime.InteropServices.Marshal.FinalReleaseComObject(shell);
        }

        private void LaunchApp()
        {
            try
            {
                string aimExe = Path.Combine(_extractPath, "AIM.exe");
                if (File.Exists(aimExe))
                {
                    Process.Start(aimExe);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Could not launch AIM: {ex.Message}", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void Log(TextBox logTextBox, string message)
        {
            if (logTextBox.InvokeRequired)
            {
                logTextBox.Invoke(new Action(() => Log(logTextBox, message)));
                return;
            }

            logTextBox.AppendText(message + Environment.NewLine);
            logTextBox.ScrollToCaret();
            Application.DoEvents();
        }
